import fs from 'node:fs';import path from 'node:path';import crypto from 'node:crypto';import assert from 'node:assert/strict';import {spawn,spawnSync} from 'node:child_process';

const json=n=>JSON.parse(fs.readFileSync(path.join(dir,n),'utf8'));
const hash=p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex');

assert.equal(gate.reviewSession,review.sessionId);assert.equal(coverage.reviewSession,review.sessionId);assert.equal(gate.verdictSHA256,hash(path.join(dir,'r-anthropic-verdict.md')));
assert.equal(coverage.rawSHA256,hash(path.join(dir,'r-anthropic-raw.jsonl')));assert.equal(coverage.complete,true);assert.equal(coverage.hashVerified,true);assert.equal(coverage.completeFiles,coverage.requiredFiles);

const active=JSON.parse(fs.readFileSync(path.join(root,'.aep-active'),'utf8'));assert.equal(active.attempt,1);assert.equal(active.max_attempts,3);assert.equal(active.track,'ops-v4-financeiro-retirada-garantia');assert.equal(active.test_command,raw.test_command);
