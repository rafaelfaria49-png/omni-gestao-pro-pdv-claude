// Prova da estratégia B: papel de runtime novo + corte do papel antigo no database de produção,
// com clientes Prisma ANTIGOS (Int, schema da main) mantendo pool aberto durante o corte.
// Banco descartável (127.0.0.1:55447). Rodar depois de 00-setup.cjs.
// Cada etapa é conferida no catálogo (has_*_privilege / pg_stat_activity), nunca pelo exit code:
// REVOKE de quem não é dono termina com WARNING e exit 0 sem revogar nada (achado da 1ª rodada).
const fs = require("node:fs")
const path = require("node:path")
const { spawnSync } = require("node:child_process")
const { psql, pw, client, errLine, DIR } = require("./lib.cjs")

const OLD = "neondb_owner_sim"
const PROD = "omnigestao_prod"
const CAND = "omnigestao_prod_candidate"
const APPLY = "C:\\Projetos\\omni-gestao\\.worktrees\\estoque-manual-kg-001\\docs\\padaria\\pesagem\\estoque-kg-migration-apply.sql"
const log = []
const out = (s) => { console.log(s); log.push(s) }
const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
const one = (s) => s.replace(/\s+/g, " ").trim()

function sql(role, db, s, label, { file = false } = {}) {
  const r = psql(role, db, s, { file })
  const warn = /WARNING|AVISO/.test(r.err)
  const ok = r.ok && !warn
  out(`  [${role}@${db}] ${label}: ${ok ? "OK" : warn ? "WARNING (tratado como falha)" : "ERRO"}${r.out ? " -> " + r.out.replace(/\n/g, " | ") : ""}${ok ? "" : " -> " + one(r.err).slice(0, 220)}`)
  return { ...r, ok }
}

async function tryOp(label, fn) {
  try { const v = await fn(); out(`  ${label}: SUCESSO${v !== undefined ? " -> " + JSON.stringify(v) : ""}`); return { ok: true, v } }
  catch (e) { out(`  ${label}: FALHOU -> ${errLine(e)}`); return { ok: false, e } }
}

const stock = () => psql("pgadmin", PROD, "SELECT string_agg(id || '=' || stock::text, ' ' ORDER BY id) FROM estoque_produtos").out
const sessoesAntigas = () => psql("pgadmin", "postgres", `SELECT count(*) FROM pg_stat_activity WHERE datname='${PROD}' AND usename='${OLD}'`).out
function matriz(label) {
  const m = psql("pgadmin", PROD, `SELECT string_agg(rolname || ': connect=' || has_database_privilege(oid, '${PROD}', 'CONNECT')
      || ' update_estoque=' || has_table_privilege(oid, 'estoque_produtos', 'UPDATE')
      || ' dono_por_heranca=' || pg_has_role(oid, (SELECT relowner FROM pg_class WHERE oid='estoque_produtos'::regclass), 'USAGE')
      || ' pg_write_all_data=' || pg_has_role(oid, 'pg_write_all_data', 'USAGE'), ' | ' ORDER BY rolname)
      FROM pg_roles WHERE rolcanlogin AND NOT rolsuper`)
  out(`  matriz ${label}: ${m.out}`)
}

async function main() {
  out(`# ESTOQUE-MANUAL-KG-001 - prova de isolamento de credenciais - ${new Date().toISOString()}`)
  out(`# PG ${psql("pgadmin", "postgres", "SHOW server_version").out} descartavel 127.0.0.1:55447 - papel antigo=${OLD} (dono de tudo + membro de neon_superuser_sim com pg_write_all_data, como neondb_owner no Neon; usado por Production, Preview, candidate e .env locais)`)
  out(`# clientes: "old" = Prisma 6.19.3 gerado do schema da main a288ea2 (Int); "new" = schema da PR #315 cdb1d19 (Float)`)

  out("\n## 0. Escritores antigos com pool aberto (codigo da main)")
  const oldA = client("old", OLD, PROD, "old-deploy-A")       // deployment antigo de produção, função quente
  const oldScript = client("old", OLD, PROD, "old-script")   // script local da main com .env antigo
  await tryOp("old-deploy-A le", () => oldA.produto.findUnique({ where: { id: "p-un-7" }, select: { stock: true } }))
  await tryOp("old-script le", () => oldScript.produto.count())
  out(`  sessoes do papel antigo em ${PROD}: ${sessoesAntigas()}`)

  out("\n## 1. Trocar so a SENHA do papel antigo nao basta")
  sql("pgadmin", "postgres", `ALTER ROLE ${OLD} PASSWORD 'rotacionada-${Date.now()}'`, "ALTER ROLE ... PASSWORD (rotacao)")
  await tryOp("old-deploy-A grava com a sessao ja autenticada (p-un-7 -> 8)", () => oldA.produto.update({ where: { id: "p-un-7" }, data: { stock: 8 } }).then((p) => p.stock))
  const fresh1 = client("old", OLD, PROD, "old-cold-start-1")
  await tryOp("deployment antigo em cold start (senha velha) conecta", () => fresh1.produto.count())
  await fresh1.$disconnect().catch(() => {})
  out(`  sessoes do papel antigo ainda abertas: ${sessoesAntigas()} (pool persistente segue gravando ate ser encerrado)`)
  sql("pgadmin", "postgres", `ALTER ROLE ${OLD} PASSWORD '${pw(OLD)}'`, "senha original restaurada (o resto da prova nao depende de senha)")

  out("\n## 2. PRE-JANELA (sem efeito nos antigos): papeis novos criados pelo papel antigo, como no Neon via SQL")
  for (const r of ["omni_owner", "omni_migrator", "omni_runtime_v2"]) psql("pgadmin", "postgres", `DROP ROLE IF EXISTS ${r}`)
  sql(OLD, CAND, "CREATE ROLE omni_owner NOLOGIN", "CREATE ROLE omni_owner NOLOGIN (dono novo dos objetos de prod)")
  sql(OLD, CAND, `CREATE ROLE omni_migrator LOGIN PASSWORD '${pw("omni_migrator")}'`, "CREATE ROLE omni_migrator LOGIN (admin/janela)")
  sql(OLD, CAND, `CREATE ROLE omni_runtime_v2 LOGIN PASSWORD '${pw("omni_runtime_v2")}'`, "CREATE ROLE omni_runtime_v2 LOGIN (runtime de Production)")
  sql(OLD, CAND, "GRANT omni_owner TO omni_migrator WITH INHERIT TRUE, SET TRUE", "GRANT omni_owner TO omni_migrator")
  fs.writeFileSync(path.join(DIR, "grants.sql"), `
GRANT CONNECT ON DATABASE ${PROD} TO omni_runtime_v2, omni_migrator;
GRANT USAGE ON SCHEMA public TO omni_runtime_v2;
GRANT SELECT, INSERT, UPDATE, DELETE ON ALL TABLES IN SCHEMA public TO omni_runtime_v2;
GRANT USAGE, SELECT, UPDATE ON ALL SEQUENCES IN SCHEMA public TO omni_runtime_v2;`)
  sql(OLD, PROD, path.join(DIR, "grants.sql"), "GRANT DML ao runtime v2 (pelo dono atual)", { file: true })
  matriz("antes do deploy")

  out("\n## 3. PRE-JANELA: deployment NOVO (runtime v2) convive com os antigos")
  const novo = client("old", "omni_runtime_v2", PROD, "new-deploy-runtime-v2") // schema igual ate a migration
  await tryOp("runtime v2 le", () => novo.produto.count())
  await tryOp("runtime v2 grava venda simulada (p-un-13 -> 12)", () => novo.produto.update({ where: { id: "p-un-13" }, data: { stock: 12 } }).then((p) => p.stock))
  await tryOp("runtime v2 tenta DDL (nao deve)", () => novo.$executeRawUnsafe("ALTER TABLE estoque_produtos ADD COLUMN x int"))
  await tryOp("old-deploy-A ainda grava (antes do corte)", () => oldA.produto.update({ where: { id: "p-un-7" }, data: { stock: 7 } }).then((p) => p.stock))

  out("\n## 4. CORTE 1/3 (janela): posse de prod -> omni_owner e papel antigo fora, NUMA transacao")
  // venda em voo de um escritor antigo, aberta antes do corte
  const emVoo = oldScript.$transaction(async (tx) => {
    await tx.produto.update({ where: { id: "p-pao" }, data: { stock: 2 } })
    await sleep(4000)
    await tx.produto.update({ where: { id: "p-un-13" }, data: { stock: 11 } })
    return "commit"
  }, { timeout: 20000 }).then((v) => ({ ok: true, v }), (e) => ({ ok: false, e }))
  await sleep(300)
  fs.writeFileSync(path.join(DIR, "cut-ownership.sql"), `
SET lock_timeout = '15s';   -- espera vendas em voo terminarem (regime integer, inofensivas); estourou = nada muda
GRANT omni_owner TO ${OLD} WITH INHERIT TRUE, SET TRUE;   -- so para poder transferir
ALTER DATABASE ${PROD} OWNER TO omni_owner;                 -- PRIMEIRO: public pertence a pg_database_owner, entao omni_owner ganha CREATE no schema (ALTER TABLE ... OWNER exige)
DO $$ DECLARE r record; BEGIN
  FOR r IN SELECT c.oid::regclass AS obj FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace
            WHERE n.nspname='public' AND c.relkind IN ('r','p','v','m','f') AND pg_get_userbyid(c.relowner)='${OLD}' LOOP
    EXECUTE format('ALTER TABLE %s OWNER TO omni_owner', r.obj);
  END LOOP;
  FOR r IN SELECT c.oid::regclass AS obj FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace
            WHERE n.nspname='public' AND c.relkind='S' AND pg_get_userbyid(c.relowner)='${OLD}' LOOP
    EXECUTE format('ALTER SEQUENCE %s OWNER TO omni_owner', r.obj);
  END LOOP;
  FOR r IN SELECT t.oid::regtype AS obj FROM pg_type t JOIN pg_namespace n ON n.oid=t.typnamespace
            WHERE n.nspname='public' AND t.typtype IN ('e','d','c') AND t.typrelid=0 AND pg_get_userbyid(t.typowner)='${OLD}' LOOP
    EXECUTE format('ALTER TYPE %s OWNER TO omni_owner', r.obj);
  END LOOP;
  FOR r IN SELECT p.oid::regprocedure AS obj FROM pg_proc p JOIN pg_namespace n ON n.oid=p.pronamespace
            WHERE n.nspname='public' AND pg_get_userbyid(p.proowner)='${OLD}' LOOP
    EXECUTE format('ALTER ROUTINE %s OWNER TO omni_owner', r.obj);
  END LOOP;
END $$;
REVOKE omni_owner FROM ${OLD};                              -- papel antigo deixa de herdar a posse
DO $$ BEGIN
  IF EXISTS (SELECT 1 FROM pg_class c JOIN pg_namespace n ON n.oid=c.relnamespace WHERE n.nspname='public' AND pg_get_userbyid(c.relowner)='${OLD}')
     OR EXISTS (SELECT 1 FROM pg_type t JOIN pg_namespace n ON n.oid=t.typnamespace WHERE n.nspname='public' AND pg_get_userbyid(t.typowner)='${OLD}' AND t.typtype IN ('e','d','c') AND t.typrelid=0)
     OR (SELECT pg_get_userbyid(datdba) FROM pg_database WHERE datname='${PROD}') <> 'omni_owner'
     OR pg_has_role('${OLD}', 'omni_owner', 'USAGE') THEN
    -- has_table_privilege NAO entra: no Neon o papel antigo herda pg_write_all_data (neon_superuser) e segue com UPDATE
    RAISE EXCEPTION 'KG_CORTE_POSSE_INCOMPLETO';
  END IF;
  RAISE NOTICE 'KG_CORTE_POSSE_OK schema_public_owner=%', pg_get_userbyid((SELECT nspowner FROM pg_namespace WHERE nspname='public'));
END $$;`)
  // spawn ASSÍNCRONO: spawnSync congela o event loop e a venda em voo nunca termina (falso lock_timeout)
  const t0 = Date.now()
  const cut = await new Promise((resolve) => {
    const ch = require("node:child_process").spawn("C:\\Program Files\\PostgreSQL\\17\\bin\\psql.exe", ["-X", "-q", "-v", "ON_ERROR_STOP=1", "-1", "-h", "127.0.0.1", "-p", "55447", "-U", OLD, "-d", PROD, "-f", path.join(DIR, "cut-ownership.sql")],
      { env: { ...process.env, PGPASSWORD: pw(OLD), PGCLIENTENCODING: "UTF8" } })
    let stderr = ""
    ch.stderr.on("data", (d) => { stderr += d })
    ch.on("close", (status) => resolve({ status, stderr }))
  })
  out(`  cut-ownership.sql como papel antigo (psql -1): exit=${cut.status} em ${Date.now() - t0} ms ${one(cut.stderr || "").replace(/^.*NOTICE:/, "NOTICE:").slice(0, 200)}`)
  out(`  dono do candidate (nao pode mudar): ${psql("pgadmin", "postgres", `SELECT pg_get_userbyid(datdba) FROM pg_database WHERE datname='${CAND}'`).out}`)
  matriz("depois da posse")
  await tryOp("old-deploy-A grava com a sessao ABERTA, logo apos a troca de posse (p-un-7 -> 6)", () => oldA.produto.update({ where: { id: "p-un-7" }, data: { stock: 6 } }).then((p) => p.stock))
  await tryOp("old-deploy-A ainda LE com a sessao aberta", () => oldA.produto.count())
  const r = await emVoo
  out(`  venda em voo (transacao aberta antes do corte): ${r.ok ? "COMMITOU -> " + r.v : "ABORTADA -> " + errLine(r.e)}`)
  out(`  sessoes do papel antigo em ${PROD}: ${sessoesAntigas()} (continuam conectadas; com pg_write_all_data a posse NAO tira a escrita)`)

  out("\n## 5. CORTE 2/3: nenhuma conexao NOVA do papel antigo (inclui pooler reconectando e cold starts)")
  sql("omni_migrator", PROD, `REVOKE CONNECT, TEMPORARY ON DATABASE ${PROD} FROM PUBLIC`, "REVOKE CONNECT, TEMPORARY FROM PUBLIC (agora como dono via omni_owner)")
  sql("omni_migrator", PROD, `REVOKE ALL ON DATABASE ${PROD} FROM ${OLD}`, "REVOKE ALL ON DATABASE FROM papel antigo")
  const fresh2 = client("old", OLD, PROD, "old-cold-start-2")
  await tryOp("deployment antigo em cold start conecta", () => fresh2.produto.count())
  await fresh2.$disconnect().catch(() => {})
  out(`  sessoes antigas ainda abertas (REVOKE CONNECT nao derruba ninguem): ${sessoesAntigas()}`)

  out("\n## 6. CORTE 3/3: encerrar as sessoes do papel antigo em prod (executado pelo proprio papel antigo, a partir do candidate)")
  sql(OLD, CAND, `SELECT 'encerradas=' || count(*) FILTER (WHERE pg_terminate_backend(pid)) FROM pg_stat_activity WHERE datname='${PROD}' AND usename='${OLD}'`, "pg_terminate_backend")
  out(`  sessoes do papel antigo em ${PROD}: ${sessoesAntigas()}`)
  for (let i = 1; i <= 3; i++) await tryOp(`old-deploy-A tentativa ${i} (Prisma reconecta sozinho)`, () => oldA.produto.update({ where: { id: "p-un-7" }, data: { stock: 5 } }).then((p) => p.stock))
  await tryOp("old-script le", () => oldScript.produto.count())
  const p1 = psql(OLD, PROD, "SELECT 1")
  out(`  psql com a credencial antiga em ${PROD}: ${p1.ok ? "CONECTOU" : "RECUSADO -> " + one(p1.err).replace(/^.*FATAL/, "FATAL")}`)
  const p2 = psql(OLD, CAND, "SELECT current_database()")
  out(`  credencial antiga no candidate (Preview/local seguem ate migrarem): ${p2.ok ? "CONECTOU -> " + p2.out : "RECUSADO"}`)
  const p3 = psql("omni_runtime_v2", CAND, "SELECT count(*) FROM estoque_produtos")
  out(`  runtime v2 no candidate: ${p3.ok ? "LEU -> " + p3.out : "SEM PRIVILEGIO -> " + one(p3.err).slice(0, 90)}`)
  await sleep(5000)
  out(`  sessoes do papel antigo em ${PROD} 5 s depois: ${sessoesAntigas()}`)
  await tryOp("runtime v2 segue gravando (p-un-13 -> 10)", () => novo.produto.update({ where: { id: "p-un-13" }, data: { stock: 10 } }).then((p) => p.stock))
  out(`  estoque real: ${stock()}`)
  matriz("depois do corte")

  out("\n## 7. Recuperacao do corte ANTES da migration (reabrir o papel antigo e cortar de novo)")
  fs.writeFileSync(path.join(DIR, "undo-cut.sql"), `
GRANT CONNECT ON DATABASE ${PROD} TO ${OLD};`)
  sql("omni_migrator", PROD, path.join(DIR, "undo-cut.sql"), "recuperacao 1/2: omni_migrator devolve CONNECT", { file: true })
  sql(OLD, CAND, `GRANT omni_owner TO ${OLD} WITH INHERIT TRUE, SET TRUE`, "recuperacao 2/2: papel antigo se re-adiciona a omni_owner (ADMIN OPTION de quem criou o papel; omni_migrator nao pode)")
  const fresh4 = client("old", OLD, PROD, "old-cold-start-4")
  await tryOp("cold start antigo depois da reversao grava (p-un-7 -> 7)", () => fresh4.produto.update({ where: { id: "p-un-7" }, data: { stock: 7 } }).then((p) => p.stock))
  await fresh4.$disconnect().catch(() => {})
  sql(OLD, PROD, `REVOKE omni_owner FROM ${OLD}`, "corte refeito: membro fora")
  sql("omni_migrator", PROD, `REVOKE ALL ON DATABASE ${PROD} FROM ${OLD}`, "corte refeito: CONNECT fora")
  sql(OLD, CAND, `SELECT 'encerradas=' || count(*) FILTER (WHERE pg_terminate_backend(pid)) FROM pg_stat_activity WHERE datname='${PROD}' AND usename='${OLD}'`, "corte refeito: sessoes")
  matriz("depois do corte refeito")

  out("\n## 8. Migration KG aplicada pelo omni_migrator com o script verificado")
  const ap = spawnSync("C:\\Program Files\\PostgreSQL\\17\\bin\\psql.exe", ["-X", "-q", "-v", "ON_ERROR_STOP=1", "-1", "-h", "127.0.0.1", "-p", "55447", "-U", "omni_migrator", "-d", PROD, "-f", APPLY],
    { env: { ...process.env, PGPASSWORD: pw("omni_migrator"), PGCLIENTENCODING: "UTF8" }, encoding: "utf8" })
  out(`  apply: exit=${ap.status} ${one(ap.stderr || "").replace(/^.*NOTICE:/, "NOTICE:").slice(0, 200)}`)
  out(`  tipos: ${psql("pgadmin", PROD, "SELECT string_agg(table_name || '.' || column_name || ':' || data_type, ' ' ORDER BY table_name, column_name) FROM information_schema.columns WHERE (table_name, column_name) IN (('estoque_produtos','stock'),('produto_depositos','quantidade'),('movimentacoes_estoque','quantidade'),('movimentacoes_estoque','estoqueAntes'),('movimentacoes_estoque','estoqueDepois'))").out}`)
  out(`  runtime v2 ainda com DML depois do ALTER TYPE: ${psql("pgadmin", PROD, "SELECT has_table_privilege('omni_runtime_v2','estoque_produtos','UPDATE')").out}`)

  out("\n## 9. Depois da migration: codigo novo grava fracao; codigo antigo nao alcanca o banco")
  await novo.$disconnect()
  const novoFloat = client("new", "omni_runtime_v2", PROD, "new-deploy-float")
  await tryOp("runtime v2 (Float) grava 3,96 kg em p-pao", () => novoFloat.produto.update({ where: { id: "p-pao" }, data: { stock: 3.96 } }).then((p) => p.stock))
  for (const [nome, c] of [["old-deploy-A", oldA], ["old-script", oldScript]]) {
    await tryOp(`${nome} (Int) tenta gravar p-un-7 -> 13`, () => c.produto.update({ where: { id: "p-un-7" }, data: { stock: 13 } }).then((p) => p.stock))
  }
  out(`  estoque real: ${stock()}`)
  out(`  linhas corrompidas (NaN, denormal ou fora de milesimos): ${psql("pgadmin", PROD, "SELECT count(*) FROM estoque_produtos WHERE stock = 'NaN' OR (stock <> 0 AND abs(stock) < 0.001) OR round(stock::numeric, 3) <> stock::numeric").out}`)
  out(`  sessoes do papel antigo em ${PROD}: ${sessoesAntigas()}`)

  await Promise.allSettled([oldA.$disconnect(), oldScript.$disconnect(), novoFloat.$disconnect()])
  fs.writeFileSync(path.join(DIR, "10-credential-isolation.txt"), log.join("\n") + "\n")
}
main().catch((e) => { console.error(e); fs.writeFileSync(path.join(DIR, "10-credential-isolation.txt"), log.join("\n") + "\nFATAL " + e + "\n"); process.exit(1) })
