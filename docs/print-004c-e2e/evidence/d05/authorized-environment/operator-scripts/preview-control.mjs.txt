function command(args, input) {

    for (const raw of [r.stdout, r.stderr]) { try { const parsed = JSON.parse(raw); const e = parsed.error ?? parsed;
      details = { code: typeof e.code === 'string' && /^[A-Za-z0-9_.-]+$/.test(e.code) ? e.code : null,
        key: typeof e.key === 'string' && keys.includes(e.key) ? e.key : null } } catch { /* discard raw data */ } }
    console.error(JSON.stringify({ event: 'D05_VERCEL_REQUEST_FAILURE', command: args[0], endpoint: args[0] === 'api' ? args[1] : 'PREVIEW_METADATA',
      exit: r.status, processError: r.error?.code ?? null, network: /ETIMEDOUT|ENOTFOUND|ECONNRESET|fetch failed/.test(r.stderr ?? ''), details,
      mentionedKeys: keys.filter(key => (r.stderr ?? '').includes(key)),
      reasonWords: ['empty','required','reserved','invalid','sensitive','encrypted','branch','target','quota','billing','plan','permission','team','project','not found','body','object','array','authentication','limit','cannot'].filter(word => (r.stderr ?? '').toLowerCase().includes(word)) }))

  }
  try { return JSON.parse(r.stdout) } catch { throw new Error('D05_VERCEL_RESPONSE_INVALID') }
}
function api(endpoint, method = 'GET', data) {
  // CLI57 parses stdin arrays but its HTTP client serializes only objects; array bodies arrive invalid at the API.
  if (Array.isArray(data)) throw new Error('D05_CLI_ARRAY_BODY_REFUSED')
  const url = endpoint + (endpoint.includes('?') ? '&' : '?') + 'teamId=' + teamId
  return command(['api', url, '--method', method, '--raw', ...(data ? ['--header', 'Content-Type: application/json', '--input', '-'] : [])], data ? JSON.stringify(data) : undefined)
}
