/**
 * D05 Preview finalization (2026-11-08). Operates ONLY on the D05 branch variables of the authorized project and on a
 * Preview deployment of the reviewed commit. Never prints values. Production, other branches and project settings: untouched.
 * Ops: auth-origin | deploy | status <dpl_id>
 */
import { spawnSync } from 'node:child_process'
import { createHash } from 'node:crypto'
import { readFileSync, writeFileSync, existsSync } from 'node:fs'
import path from 'node:path'
import { savePrivateEnvironment } from '../environment.mjs'
import { D05_TARGET, d05IsolationErrors } from '../../../lib/printing/direct/homologation.mjs'

const sha = '1c39fcd2e06a434f8e1dd49b67ef49f17b0bacbf'
const teamId = 'team_IInBSQbKO8P5XpAsvz16Kc3g', scope = 'rafaelfaria49-4373s-projects'
const cli = 'C:/Users/rafae/AppData/Roaming/npm/node_modules/vercel/dist/index.js'
const privateRoot = 'C:/Users/rafae/.codex/visualizations/2026/11/05/01a1ac95-8f12-7141-9a0c-f878d5bf822f/d05-private'
const out = 'docs/print-004c-e2e/evidence/d05/authorized-environment/'
const hash = data => createHash('sha256').update(JSON.stringify(data)).digest('hex')
const save = (name, value) => writeFileSync(out + name + '.json', JSON.stringify(value, null, 2) + '\n')
const AUTH_KEYS = ['AUTH_URL', 'NEXTAUTH_URL']

function command(args, input) {
  const r = spawnSync(process.execPath, [cli, ...args, '--scope', scope], { input, encoding: 'utf8', windowsHide: true,
    timeout: 120000, maxBuffer: 32 << 20, env: { ...process.env, NO_UPDATE_NOTIFIER: '1', VERCEL_TELEMETRY_DISABLED: '1' } })
  if (r.status !== 0 || r.error) {
    let code = null
    for (const raw of [r.stdout, r.stderr]) { try { const e = JSON.parse(raw).error ?? {}; if (typeof e.code === 'string' && /^[A-Za-z0-9_.-]+$/.test(e.code)) code = e.code } catch { /* discard */ } }
    console.error(JSON.stringify({ event: 'D05_VERCEL_REQUEST_FAILURE', command: args[0], exit: r.status, code }))
    throw new Error('D05_VERCEL_REQUEST_FAILED')
  }
  try { return JSON.parse(r.stdout) } catch { throw new Error('D05_VERCEL_RESPONSE_INVALID') }
}
function api(endpoint, method = 'GET', data) {
  const url = endpoint + (endpoint.includes('?') ? '&' : '?') + 'teamId=' + teamId
  return command(['api', url, '--method', method, '--raw', ...(data ? ['--header', 'Content-Type: application/json', '--input', '-'] : [])], data ? JSON.stringify(data) : undefined)
}
function checkedProject() {
  const p = api('/v9/projects/' + D05_TARGET.projectId), team = api('/v2/teams/' + teamId)
  if (p.id !== D05_TARGET.projectId || p.name !== 'omni-gestao-pro' || p.accountId !== teamId || p.link?.type !== 'github' ||
      p.link.org !== 'rafaelfaria49-png' || p.link.repo !== 'omni-gestao-pro-pdv-claude' || Number(p.link.repoId) !== 1208557742 ||
      team.id !== teamId || team.billing?.plan !== 'hobby' || p.ssoProtection || p.passwordProtection) throw new Error('D05_VERCEL_TARGET_OR_COST_OR_ACCESS_REFUSED')
}
/** Metadata only (values are discarded here, never returned). */
function metadata() {
  const data = api('/v10/projects/' + D05_TARGET.projectId + '/env?target=preview')
  return data.envs.map(e => ({ id: e.id, key: e.key, type: e.type, target: [...(e.target ?? [])].sort(), gitBranch: e.gitBranch ?? null, updatedAt: e.updatedAt ?? null }))
    .sort((a, b) => String(a.id).localeCompare(String(b.id)))
}
const others = rows => rows.filter(e => e.gitBranch !== D05_TARGET.branch)
function gate() {
  const proof = JSON.parse(readFileSync(out + 'ACTIVATION_GATES_' + sha.slice(0, 7) + '.json', 'utf8'))
  if (proof.commit !== sha || !['APPROVE', 'APPROVE_WITH_NITS'].includes(proof.independentReview) || proof.runs?.length !== 4 || proof.runs.some(r => r.headSha !== sha || r.conclusion !== 'success')) throw new Error('D05_REVIEW_CI_GATE_REQUIRED')
  for (const run of proof.runs) {
    const r = spawnSync('gh', ['api', 'repos/rafaelfaria49-png/omni-gestao-pro-pdv-claude/actions/runs/' + run.id], { encoding: 'utf8', windowsHide: true, timeout: 30000 })
    const a = r.status === 0 ? JSON.parse(r.stdout) : null
    if (!a || a.head_sha !== sha || a.head_branch !== D05_TARGET.branch || a.conclusion !== 'success') throw new Error('D05_CURRENT_CI_NOT_GREEN')
  }
  const d = spawnSync('git', ['diff', '--quiet', sha, '--', 'lib/printing/direct', 'tools/print-d02', 'tools/print-d03', 'tools/print-d04', 'tools/omni-print-agent', 'vercel.json', '.github/workflows', 'scripts/vercel-build.mjs', 'tools/print-d05/build-preview.mjs'], { windowsHide: true })
  if (d.status !== 0) throw new Error('D05_REVIEWED_SOURCE_CHANGED')
}
function readRawPrivate() {
  const ps = path.join(process.env.SystemRoot || 'C:/Windows', 'System32/WindowsPowerShell/v1.0/powershell.exe')
  const r = spawnSync(ps, ['-NoProfile', '-NonInteractive', '-File', path.resolve('tools/print-d05/private-state.ps1'), '-Mode', 'Read', '-PrivateRoot', privateRoot, '-InternalPipe'], { encoding: 'utf8', windowsHide: true, timeout: 30000 })
  if (r.status !== 0 || r.error) throw new Error('PRIVATE_STATE_UNAVAILABLE')
  return JSON.parse(r.stdout)
}
try {
  const [operation, argument] = process.argv.slice(2)
  if (operation === 'auth-origin') {
    gate(); checkedProject()
    const values = readRawPrivate()
    for (const k of AUTH_KEYS) values[k] = D05_TARGET.origin
    const env = { ...values, VERCEL: '1', VERCEL_ENV: 'preview', VERCEL_TARGET_ENV: 'preview', VERCEL_PROJECT_ID: D05_TARGET.projectId, VERCEL_GIT_COMMIT_REF: D05_TARGET.branch }
    if (d05IsolationErrors(env).length) throw new Error('D05_PRIVATE_STATE_ISOLATION_REFUSED')
    const before = metadata()
    const owned = before.filter(e => e.gitBranch === D05_TARGET.branch && AUTH_KEYS.includes(e.key))
    if (owned.length !== 2 || owned.some(e => e.target.length !== 1 || e.target[0] !== 'preview')) throw new Error('D05_AUTH_BRANCH_VARIABLES_NOT_FOUND')
    savePrivateEnvironment(privateRoot, values)
    for (const e of owned) {
      const r = api('/v9/projects/' + D05_TARGET.projectId + '/env/' + e.id, 'PATCH', { value: D05_TARGET.origin, type: 'encrypted', target: ['preview'], gitBranch: D05_TARGET.branch })
      if (r.id !== e.id || r.key !== e.key || r.gitBranch !== D05_TARGET.branch) throw new Error('D05_AUTH_PATCH_UNEXPECTED')
      console.log('D05_AUTH_ORIGIN_SET ' + e.key)
    }
    const after = metadata()
    if (hash(others(before)) !== hash(others(after))) throw new Error('D05_NON_BRANCH_METADATA_CHANGED_STOP')
    save('PREVIEW_AUTH_ORIGIN_SET', { at: new Date().toISOString(), commit: sha, projectId: D05_TARGET.projectId, branch: D05_TARGET.branch,
      keys: AUTH_KEYS, origin: D05_TARGET.origin, variableIds: owned.map(e => ({ key: e.key, id: e.id })), privateStateUpdated: true,
      nonD05Fingerprint: hash(others(after)), nonD05MetadataUnchanged: true, directTransportEnabled: false, valuesNotEmitted: true })
  } else if (operation === 'deploy') {
    gate(); checkedProject()
    if (!existsSync(out + 'PREVIEW_AUTH_ORIGIN_SET.json')) throw new Error('D05_AUTH_ORIGIN_REQUIRED')
    const flag = api('/v10/projects/' + D05_TARGET.projectId + '/env?target=preview&decrypt=false').envs.find(e => e.gitBranch === D05_TARGET.branch && e.key === 'OMNI_PRINT_DIRECT_ENABLED')
    if (!flag || flag.type !== 'plain' || flag.value !== '0') throw new Error('D05_DIRECT_TRANSPORT_MUST_STAY_DISABLED')
    const p = api('/v13/deployments', 'POST', { name: 'omni-gestao-pro', project: D05_TARGET.projectId, gitSource: { type: 'github', repoId: 1208557742, ref: D05_TARGET.branch, sha } })
    if (p.target && p.target !== 'preview') throw new Error('D05_DEPLOYMENT_TARGET_REFUSED')
    save('PREVIEW_DEPLOYMENT_CREATED_' + sha.slice(0, 7), { at: new Date().toISOString(), commit: sha, projectId: D05_TARGET.projectId, branch: D05_TARGET.branch,
      id: p.id, url: p.url, target: p.target ?? 'preview', readyState: p.readyState, directTransportEnabled: false, supersedes: 'dpl_B9vJfGhLWMuaJPqLkLArFyxzXmZa' })
    console.log(JSON.stringify({ id: p.id, url: p.url, state: p.readyState }))
  } else if (operation === 'status') {
    if (!/^dpl_[A-Za-z0-9]+$/.test(argument ?? '')) throw new Error('D05_DEPLOYMENT_ID_REQUIRED')
    const p = api('/v13/deployments/' + argument)
    if (p.projectId !== D05_TARGET.projectId || (p.target && p.target !== 'preview') || p.meta?.githubCommitRef !== D05_TARGET.branch || p.meta?.githubCommitSha !== sha) throw new Error('D05_DEPLOYMENT_SCOPE_MISMATCH')
    const result = { at: new Date().toISOString(), projectId: p.projectId, id: p.id, url: p.url, target: p.target ?? 'preview', state: p.readyState,
      gitRef: p.meta.githubCommitRef, sha: p.meta.githubCommitSha, errorCode: p.errorCode ?? null }
    save('PREVIEW_DEPLOYMENT_DISABLED_STATUS_' + sha.slice(0, 7), result); console.log(JSON.stringify(result))
  } else throw new Error('D05_PREVIEW_OPERATION_REQUIRED')
} catch (e) { console.error(e instanceof Error && /^[A-Z0-9_]+$/.test(e.message) ? e.message : 'D05_PREVIEW_OPERATION_FAILED'); process.exitCode = 1 }
