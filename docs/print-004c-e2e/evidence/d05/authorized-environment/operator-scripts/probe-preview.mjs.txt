/** Live D05 synthetic login/permissions only. No pairing, sale, job, printer or credentials in output. */
import { readFileSync, writeFileSync } from 'node:fs'
import { privateEnvironment } from '../environment.mjs'
import { D05_TARGET } from '../../../lib/printing/direct/homologation.mjs'
const out = 'docs/print-004c-e2e/evidence/d05/authorized-environment/'
const sha = '172882c6e721bee506307c5ec2aac50ec2fca9ef'
const mode = process.argv[2]
const reportFile = mode === 'disabled' ? 'PREVIEW_PROBE_DISABLED.json' : mode === 'enabled' ? 'PREVIEW_PROBE_ENABLED.json' : null
const report = { at: new Date().toISOString(), commit: sha, mode, checks: [], status: 'FAIL', productionAccess: false, physicalHomologation: 'PENDING' }
let phase = 'SCOPE'
try {
  if (!['disabled', 'enabled'].includes(mode) || process.env.NODE_TLS_REJECT_UNAUTHORIZED === '0') throw new Error()
  const deployment = JSON.parse(readFileSync(out + 'PREVIEW_DEPLOYMENT_' + mode.toUpperCase() + '_STATUS.json', 'utf8'))
  if (deployment.sha !== sha || deployment.projectId !== D05_TARGET.projectId || deployment.gitRef !== D05_TARGET.branch || deployment.state !== 'READY' || deployment.target !== 'preview') throw new Error()
  const origin = new URL('https://' + deployment.url)
  if (!origin.hostname.endsWith('.vercel.app') || origin.username || origin.password || origin.search || origin.hash) throw new Error()
  report.deploymentId = deployment.id; report.url = origin.origin
  const values = privateEnvironment('C:/Users/rafae/.codex/visualizations/2026/11/05/01a1ac95-8f12-7141-9a0c-f878d5bf822f/d05-private')
  const request = async (pathname, options = {}, jar = new Map()) => {
    const url = new URL(pathname, origin)
    if (url.origin !== origin.origin) throw new Error()
    const response = await fetch(url, { ...options, redirect: 'manual', signal: AbortSignal.timeout(30000),
      headers: { cookie: [...jar].map(([k, v]) => k + '=' + v).join('; '), ...options.headers } })
    for (const line of response.headers.getSetCookie()) {
      const value = line.split(';', 1)[0], i = value.indexOf('=')
      if (i > 0) jar.set(value.slice(0, i), value.slice(i + 1))
    }
    return response
  }
  for (const [label, email, password, id] of [['ADMIN', 'admin-d05@example.invalid', values.ADMIN_DEFAULT_PASSWORD, 'd05-admin'], ['CAIXA', 'operador-d05@example.invalid', values.D05_OPERATOR_PASSWORD, 'd05-operator']]) {
    const jar = new Map()
    phase = label + '_CSRF'
    const csrf = await request('/api/auth/csrf', {}, jar), token = await csrf.json()
    if (csrf.status !== 200 || typeof token.csrfToken !== 'string') throw new Error()
    phase = label + '_LOGIN'
    const login = await request('/api/auth/callback/credentials', { method: 'POST', headers: { 'content-type': 'application/x-www-form-urlencoded', origin: origin.origin, 'X-Auth-Return-Redirect': '1' },
      body: new URLSearchParams({ csrfToken: token.csrfToken, email, password, callbackUrl: origin.origin + '/dashboard' }) }, jar)
    if (![200, 302].includes(login.status)) throw new Error()
    phase = label + '_SESSION'
    const sessionResponse = await request('/api/auth/session', {}, jar), session = await sessionResponse.json()
    if (sessionResponse.status !== 200 || session.user?.id !== id || session.user?.role !== label || JSON.stringify(session.user.allowedStoreIds) !== JSON.stringify(['d05-synthetic-store'])) throw new Error()
    report.checks.push({ check: label + '_SYNTHETIC_LOGIN', status: 'PASS', userId: id, role: session.user.role, allowedStoreIds: session.user.allowedStoreIds })
    phase = label + '_STORES'
    const storesResponse = await request('/api/stores', {}, jar), stores = await storesResponse.json()
    if (storesResponse.status !== 200 || stores.stores?.length !== 1 || stores.stores[0].id !== 'd05-synthetic-store') throw new Error()
    report.checks.push({ check: label + '_ONLY_SYNTHETIC_STORE_VISIBLE', status: 'PASS' })
    phase = label + '_PRINT_AUTHORITY'
    const admin = await request('/api/print/direct/admin', { headers: { 'x-assistec-loja-id': 'd05-synthetic-store' } }, jar), result = await admin.json()
    if (mode === 'disabled') {
      if (admin.status !== 404 || result.error !== 'DIRECT_PRINT_DISABLED') throw new Error()
    } else if (label === 'ADMIN') {
      if (admin.status !== 200 || result.ok !== true || !Array.isArray(result.installations) || result.installations.length !== 0) throw new Error()
    } else if (admin.status !== 403) throw new Error()
    report.checks.push({ check: label + '_PRINT_AUTHORITY', status: 'PASS', httpStatus: admin.status, pairedInstallations: mode === 'enabled' && label === 'ADMIN' ? 0 : undefined })
    jar.clear()
  }
  phase = 'AGENT_UNAUTHENTICATED'
  const response = await request('/api/print/agent/heartbeat', { method: 'POST', headers: { 'content-type': 'application/json' }, body: '{}' })
  if (response.status !== (mode === 'enabled' ? 401 : 404)) throw new Error()
  report.checks.push({ check: 'HTTPS_AGENT_REQUIRES_CREDENTIAL', status: 'PASS', httpStatus: response.status, redirectsFollowed: false, tlsVerificationDisabled: false })
  report.status = 'PASS'
  console.log('D05_PREVIEW_' + mode.toUpperCase() + '_PROBE_PASS')
} catch { report.failedPhase = phase; console.error('D05_PREVIEW_PROBE_FAILED_' + phase); process.exitCode = 1 }
finally { if (reportFile) writeFileSync(out + reportFile, JSON.stringify(report, null, 2) + '\n') }
